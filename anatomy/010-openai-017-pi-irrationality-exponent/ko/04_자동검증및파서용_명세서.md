sha256sum -c proofs/checksums.sha256

# 3. 파이썬 기호·수치 검증기 실행
python3 -B proofs/verify_symbolic_pi017.py

# 4. 통합 검증 게이트 (가설 변이 315/315 + Elab MetaM DoF 75/75 + nonlin 39 + oai_ref 60/60 100%)
python3 -B ../../tools/verify_all.py --root .
